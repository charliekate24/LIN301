Part 1 Write your Functions

def is_vowel(letter):
    vowels = "aeiouāēīōū"
    return letter in vowels
def cv_pattern(word):
    pattern = ""
    for letter in word:
        if is_vowel(letter):
            pattern += "V"
        else:
            pattern += "C"
    return pattern 
def syllabify(word):
    syllables = []
    current = ""
    for letter in word:
        current += letter
        if is_vowel(letter):
            syllables.append(current)
            current = ""
    return syllables
def count_syllables(word):
    return len(syllabify(word))
