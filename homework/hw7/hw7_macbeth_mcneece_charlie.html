parts = ["NOUN", "VERB", "ADJ"]

for language, df in [
    ("English", df_en),
    ("Finnish", df_fi),
    ("Greek", df_el),
]:
    words = df[df["pos"] != "PUNCT"]
    total = len(words)

    print(language)

    for pos in parts:
        count = len(words[words["pos"] == pos])
        rate = round(count / total * 1000, 2)
        print(pos, rate)

    print()

Part 2: Investigate (Scripts)

import unicodedata

for language, df in [
    ("English", df_en),
    ("Finnish", df_fi),
    ("Greek", df_el),
]:
    scripts = []

    for word in df["word"]:
        if word and word[0].isalpha():
            script = unicodedata.name(word[0]).split()[0]
            scripts.append(script)

    counts = pd.Series(scripts).value_counts()

    print(language)
    print(counts)
    print()
